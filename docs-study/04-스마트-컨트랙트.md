## 핵심 컨트랙트

### 1. ZkDaiBase.sol

모든 기본 상태와 매핑을 정의합니다.

```solidity
// 노트 상태
enum NoteState { Invalid, Valid, Trading, Spent }
mapping(bytes32 => NoteState) public notes;

// 암호화된 노트 데이터 (ECDH)
mapping(bytes32 => bytes) public encryptedNotes;

// 토큰 관리
IERC20 public dai;
uint256 constant ETH_TOKEN_TYPE = 0;
uint256 constant DAI_TOKEN_TYPE = 1;

// 이벤트
event NoteStateChange(bytes32 note, NoteState state);
```

### 주요 상태 변수

| 변수 | 타입 | 설명 |
|------|------|------|
| `notes` | mapping | noteHash → 상태 |
| `encryptedNotes` | mapping | noteHash → 암호화된 바이트 |
| `dai` | IERC20 | DAI 토큰 컨트랙트 주소 |

### 2. ZkDai.sol (노트 관리)

`mint`, `spend`, `liquidate` 기능을 제공합니다.

#### mint() - 노트 생성

```solidity
function mint(
    uint256[2] memory a,
    uint256[2][2] memory b,
    uint256[2] memory c,
    uint256[4] memory input,    // [output, noteHash, value, tokenType]
    bytes memory encryptedNote
) external payable {
    // 1. 증명 검증
    require(mintBurnVerifier.verifyProof(a, b, c, input));

    // 2. 금액 일치 확인
    if (input[3] == ETH_TOKEN_TYPE) {
        require(msg.value == input[2]);
    } else {
        dai.transferFrom(msg.sender, address(this), input[2]);
    }

    // 3. 노트 상태 변경
    notes[bytes32(input[1])] = NoteState.Valid;

    // 4. 암호화된 노트 저장
    encryptedNotes[bytes32(input[1])] = encryptedNote;
}
```

#### spend() - 노트 전송

```solidity
function spend(
    uint256[2] memory a,
    uint256[2][2] memory b,
    uint256[2] memory c,
    uint256[5] memory input,    // [output, o0Hash, o1Hash, newHash, changeHash]
    bytes memory encNote1,
    bytes memory encNote2
) external {
    // 1. 증명 검증
    require(transferVerifier.verifyProof(a, b, c, input));

    // 2. 입력 노트 소비
    require(notes[bytes32(input[1])] == NoteState.Valid);
    notes[bytes32(input[1])] = NoteState.Spent;

    // EMPTY_NOTE가 아니면 두 번째 노트도 소비
    if (input[2] != EMPTY_NOTE_HASH) {
        require(notes[bytes32(input[2])] == NoteState.Valid);
        notes[bytes32(input[2])] = NoteState.Spent;
    }

    // 3. 출력 노트 생성
    notes[bytes32(input[3])] = NoteState.Valid;
    notes[bytes32(input[4])] = NoteState.Valid;
}
```

#### liquidate() - 노트 소각 (출금)

```solidity
function liquidate(
    address to,
    uint256[2] memory a,
    uint256[2][2] memory b,
    uint256[2] memory c,
    uint256[4] memory input     // [output, noteHash, value, tokenType]
) external {
    // 1. 증명 검증
    require(mintBurnVerifier.verifyProof(a, b, c, input));

    // 2. 노트 소비
    require(notes[bytes32(input[1])] == NoteState.Valid);
    notes[bytes32(input[1])] = NoteState.Spent;

    // 3. 토큰 전송
    if (input[3] == ETH_TOKEN_TYPE) {
        payable(to).transfer(input[2]);
    } else {
        dai.transfer(to, input[2]);
    }
}
```

### 3. ZkDex.sol (거래 기능)

주문 생성, 수락, 체결 기능을 제공합니다.

#### Order 구조체

```solidity
struct Order {
    bytes32 makerViewingKey;    // 메이커의 뷰잉 키
    bytes32 makerNote;          // 메이커 노트 해시
    uint256 sourceToken;        // 메이커가 제공하는 토큰
    uint256 targetToken;        // 메이커가 원하는 토큰
    uint256 price;              // 교환 비율 (10^18 기준)
    bytes32 takerNoteToMaker;   // 테이커 스테이크 노트
    bytes32 parentNote;         // 테이커 부모 노트
    OrderState state;           // Created → Taken → Settled
}

mapping(uint256 => Order) public orders;
uint256 public orderCount;
```

#### makeOrder() - 주문 생성

```solidity
function makeOrder(
    bytes32 makerViewingKey,
    uint256 targetToken,
    uint256 price,
    uint256[2] memory a,
    uint256[2][2] memory b,
    uint256[2] memory c,
    uint256[3] memory input     // [output, noteHash, tokenType]
) external returns (uint256 orderId) {
    // 1. 증명 검증
    require(makeOrderVerifier.verifyProof(a, b, c, input));

    // 2. 노트 잠금
    require(notes[bytes32(input[1])] == NoteState.Valid);
    notes[bytes32(input[1])] = NoteState.Trading;

    // 3. 주문 생성
    orderId = orderCount++;
    orders[orderId] = Order({
        makerViewingKey: makerViewingKey,
        makerNote: bytes32(input[1]),
        sourceToken: input[2],
        targetToken: targetToken,
        price: price,
        takerNoteToMaker: bytes32(0),
        parentNote: bytes32(0),
        state: OrderState.Created
    });
}
```

#### takeOrder() - 주문 수락

```solidity
function takeOrder(
    uint256 orderId,
    uint256[2] memory a,
    uint256[2][2] memory b,
    uint256[2] memory c,
    uint256[7] memory input,    // [output, parentHash, parentType, stakeHash, owner0, owner1, stakeType]
    bytes memory encryptedStakingNote
) external {
    Order storage order = orders[orderId];

    // 1. 주문 상태 확인
    require(order.state == OrderState.Created);

    // 2. 토큰 타입 일치 확인
    require(order.targetToken == input[2]);
    require(order.targetToken == input[6]);

    // 3. 스마트 노트 소유자 검증 (makerNote에서 파생)
    require(uint256(order.makerNote) >> 128 == input[4]);
    require(uint256(order.makerNote) & ((1 << 128) - 1) == input[5]);

    // 4. 증명 검증
    require(takeOrderVerifier.verifyProof(a, b, c, input));

    // 5. 노트 상태 변경
    notes[bytes32(input[1])] = NoteState.Trading;  // 부모 노트
    notes[bytes32(input[3])] = NoteState.Trading;  // 스테이크 노트

    // 6. 주문 업데이트
    order.takerNoteToMaker = bytes32(input[3]);
    order.parentNote = bytes32(input[1]);
    order.state = OrderState.Taken;
}
```

#### settleOrder() - 주문 체결

```solidity
function settleOrder(
    uint256 orderId,
    uint256[2] memory a,
    uint256[2][2] memory b,
    uint256[2] memory c,
    uint256[16] memory input,
    bytes[3] memory encDatas   // [rewardEnc, paymentEnc, changeEnc]
) external {
    Order storage order = orders[orderId];

    // 1. 주문 상태 확인
    require(order.state == OrderState.Taken);

    // 2. 주문 데이터 일치 확인
    require(bytes32(input[1]) == order.makerNote);
    require(bytes32(input[3]) == order.takerNoteToMaker);
    require(input[15] == order.price);

    // 3. 출력 노트 소유자 검증
    // 보상 노트 → parentNote에서 파생
    require(input[6] == uint256(order.parentNote) >> 128);
    require(input[7] == uint256(order.parentNote) & MASK_128);
    // 지불 노트 → makerNote에서 파생
    require(input[10] == uint256(order.makerNote) >> 128);
    require(input[11] == uint256(order.makerNote) & MASK_128);

    // 4. 증명 검증
    require(settleOrderVerifier.verifyProof(a, b, c, input));

    // 5. 입력 노트 소비
    notes[order.makerNote] = NoteState.Spent;
    notes[order.parentNote] = NoteState.Spent;
    notes[order.takerNoteToMaker] = NoteState.Spent;

    // 6. 출력 노트 생성
    notes[bytes32(input[5])] = NoteState.Valid;   // 보상
    notes[bytes32(input[9])] = NoteState.Valid;   // 지불
    notes[bytes32(input[13])] = NoteState.Valid;  // 잔돈

    // 7. 주문 완료
    order.state = OrderState.Settled;
}
```

#### convertNote() - 스마트노트 변환

```solidity
function convertNote(
    uint256[2] memory a,
    uint256[2][2] memory b,
    uint256[2] memory c,
    uint256[4] memory input,    // [output, smartHash, originHash, newHash]
    bytes memory encryptedNote
) external {
    // 1. 증명 검증
    require(convertVerifier.verifyProof(a, b, c, input));

    // 2. 스마트 노트 무효화
    require(notes[bytes32(input[1])] == NoteState.Valid);
    notes[bytes32(input[1])] = NoteState.Invalid;

    // 3. 새 일반 노트 생성
    notes[bytes32(input[3])] = NoteState.Valid;
    encryptedNotes[bytes32(input[3])] = encryptedNote;
}
```

## Groth16 검증자 컨트랙트

snarkjs가 자동 생성합니다. `contracts/verifiers/` 위치.

```solidity
// 예: MintBurnNoteVerifier.sol
contract MintBurnNoteVerifier {
    function verifyProof(
        uint256[2] memory a,      // G1 포인트
        uint256[2][2] memory b,   // G2 포인트
        uint256[2] memory c,      // G1 포인트
        uint256[4] memory input   // 공개 입력
    ) public view returns (bool);
}
```

| 검증자 | 공개 입력 수 |
|--------|-------------|
| MintBurnNoteVerifier | 4 |
| TransferNoteVerifier | 5 |
| MakeOrderVerifier | 3 |
| TakeOrderVerifier | 7 |
| SettleOrderVerifier | 16 |
| ConvertNoteVerifier | 4 |

  Invalid ──mint──▶ Valid ──spend/liquidate──▶ Spent

  Created ──takeOrder──▶ Taken ──settleOrder──▶ Settled
